cmake_minimum_required(VERSION 3.16)
project(FoxxyWiiU360 VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(foxxy_core src/memory_bus.cpp src/emulator.cpp)
target_include_directories(foxxy_core PUBLIC include)
target_compile_options(foxxy_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(foxxy_tests tests/test_main.cpp)
target_link_libraries(foxxy_tests PRIVATE foxxy_core)
enable_testing()
add_test(NAME foxxy_tests COMMAND foxxy_tests)
